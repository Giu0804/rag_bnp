import ast
import numpy as np

def parse_citation_ids(val):
    if val is None:
        return set()
    if isinstance(val, str):
        try:
            parsed = ast.literal_eval(val)
            if isinstance(parsed, list):
                return set(int(x) for x in parsed)
            elif isinstance(parsed, (int, float)):
                return {int(parsed)}
        except:
            return set()
    if isinstance(val, (list, np.ndarray, set)):
        return set(int(x) for x in val)
    return set()

def evaluate_chunk_classification_metrics(df, gt_col='preannotation', pred_col='inline_processed'):
    """
    Évalue la classification (Cité vs Non-Cité) de CHAQUE chunk du contexte
    pour CHAQUE snippet individuel.
    """
    total_snippets = 0
    total_chunks_evaluated = 0
    
    total_tp = 0
    total_fp = 0
    total_fn = 0
    total_tn = 0

    sum_snippet_recall = 0.0
    sum_snippet_precision = 0.0
    sum_snippet_f1 = 0.0
    count_exact_match = 0

    for idx, row in df.iterrows():
        # 1. Récupération de TOUS les chunks disponibles dans le contexte
        context_doc_ids = set(int(ctx["id"]) for ctx in row["contexts"])
        num_docs_in_context = len(context_doc_ids)

        # 2. Parsing du GT et des Prédictions
        gt_raw = row[gt_col]
        pred_raw = row[pred_col]

        gt_map = {}
        for s in (gt_raw if isinstance(gt_raw, list) else []):
            if isinstance(s, dict) and 'snippet_id' in s:
                pred_key = 'prediction' if 'prediction' in s else 'citations'
                gt_map[int(s['snippet_id'])] = parse_citation_ids(s.get(pred_key, []))

        pred_map = {}
        for s in (pred_raw if isinstance(pred_raw, list) else []):
            if isinstance(s, dict) and 'snippet_id' in s:
                pred_key = 'citations' if 'citations' in s else 'prediction'
                pred_map[int(s['snippet_id'])] = parse_citation_ids(s.get(pred_key, []))

        all_snippet_ids = set(gt_map.keys()) | set(pred_map.keys())

        # 3. Évaluation snippet par snippet
        for sid in all_snippet_ids:
            total_snippets += 1
            total_chunks_evaluated += num_docs_in_context

            gt_set = gt_map.get(sid, set())
            pred_set = pred_map.get(sid, set())

            # Décompte exact des 4 quadrants pour ce snippet
            tp = len(gt_set & pred_set)
            fp = len(pred_set - gt_set)
            fn = len(gt_set - pred_set)
            tn = len(context_doc_ids - (gt_set | pred_set))

            total_tp += tp
            total_fp += fp
            total_fn += fn
            total_tn += tn

            # Métriques locales du snippet
            if len(gt_set) == 0 and len(pred_set) == 0:
                rec, prec, f1 = 1.0, 1.0, 1.0
                is_exact = True
            else:
                prec = tp / (tp + fp) if (tp + fp) > 0 else 0.0
                rec = tp / (tp + fn) if (tp + fn) > 0 else 0.0
                f1 = (2 * prec * rec) / (prec + rec) if (prec + rec) > 0 else 0.0
                is_exact = (gt_set == pred_set)

            sum_snippet_recall += rec
            sum_snippet_precision += prec
            sum_snippet_f1 += f1
            if is_exact:
                count_exact_match += 1

    # Métriques globales
    chunk_accuracy = (total_tp + total_tn) / total_chunks_evaluated if total_chunks_evaluated > 0 else 0.0
    
    print("\n" + "="*85)
    print("📊 BILAN DE LA CLASSIFICATION DES CHUNKS (PAR SNIPPET)")
    print("="*85)
    print(f"Total Snippets analysés       : {total_snippets}")
    print(f"Total Décisions Chunks (Docs) : {total_chunks_evaluated}")
    print(f"Chunks BIEN classifiés (TP+TN): {total_tp + total_tn} ({chunk_accuracy*100:.2f}%)")
    print(f"  ├─ Vrais Positifs (TP)      : {total_tp} (Citations bien attribuées)")
    print(f"  └─ Vrais Négatifs (TN)      : {total_tn} (Docs hors-sujet bien ignorés)")
    print(f"Chunks MAL classifiés (FP+FN) : {total_fp + total_fn}")
    print(f"  ├─ Faux Positifs  (FP)      : {total_fp} (Sur-citations inutiles)")
    print(f"  └─ Faux Négatifs  (FN)      : {total_fn} (Citations manquées/oubliées)")
    print("-" * 85)
    print(f"Moyenne Rappel (Recall)       : {(sum_snippet_recall / total_snippets)*100:.2f}%")
    print(f"Moyenne Précision (Precision) : {(sum_snippet_precision / total_snippets)*100:.2f}%")
    print(f"Moyenne F1 Score              : {(sum_snippet_f1 / total_snippets)*100:.2f}%")
    print(f"Exact Match Snippets          : {(count_exact_match / total_snippets)*100:.2f}%")
    print("="*85)

    return {
        "tp": total_tp, "fp": total_fp, "fn": total_fn, "tn": total_tn,
        "chunk_accuracy": chunk_accuracy,
        "mean_recall": (sum_snippet_recall / total_snippets) * 100,
        "mean_precision": (sum_snippet_precision / total_snippets) * 100,
        "mean_f1": (sum_snippet_f1 / total_snippets) * 100
    }

        [f"Faux Positifs (FP)\n{fp}\n({(fp/total_decisions)*100:.1f}%)",

    ])

        yticklabels=["Attendu GT (1)", "Non-Attendu GT (0)"]
    )

    plt.title(

def run_lambda_grid_search(
    df,
    gt_col="preannotation",
    stat_mode="mean",
    lambda_start=0.1,
    lambda_end=0.9,
    step=0.05,
    gamma=0.85,
):
  lambdas = np.arange(lambda_start, lambda_end + 1e-5, step)
  results = []

  print(f"\n{'='*95}")
  print(
      f"GRID SEARCH SUR LAMBDA (Mode Stat = '{stat_mode}' | Gamma = {gamma})"
  )
  print(f"{'='*95}")
  print(
      f"{'Lambda':<8} | {'Rappel %':<10} | {'Précision %':<12} | {'F1 %':<8} |"
      f" {'Exact %':<9} | {'TP':<6} | {'FP':<6} | {'FN':<6} | {'TN':<6}"
  )
  print(f"{'-'*95}")

  for l_val in lambdas:
    l_val = round(l_val, 3)
    temp_col = []

    for _, row in df.iterrows():
      scores = compute_scores_for_sample(row, lambda_val=l_val)
      enriched = apply_adaptive_validation(
          row, scores, stat_mode=stat_mode, gamma=gamma
      )
      temp_col.append(enriched)

    df_temp = df.copy()
    df_temp["temp_eval_col"] = temp_col

    # Appel de la nouvelle fonction d'évaluation
    m = evaluate_chunk_classification_metrics(
        df_temp, gt_col=gt_col, pred_col="temp_eval_col"
    )
    m["lambda"] = l_val
    results.append(m)

    print(
        f"{l_val:<8.2f} | {m['mean_recall']:<10.2f} |"
        f" {m['mean_precision']:<12.2f} | {m['mean_f1']:<8.2f} |"
        f" {m['chunk_accuracy']*100:<9.2f} | {m['tp']:<6} | {m['fp']:<6} |"
        f" {m['fn']:<6} | {m['tn']:<6}"
    )

  print(f"{'='*95}\n")
  res_df = pd.DataFrame(results)

  # Tracé des courbes
  plt.figure(figsize=(10, 6))
  plt.plot(
      res_df["lambda"],
      res_df["mean_recall"],
      marker="o",
      label="Moyenne Rappel",
      color="#2ca02c",
      linewidth=2.5,
  )
  plt.plot(
      res_df["lambda"],
      res_df["mean_precision"],
      marker="s",
      label="Moyenne Précision",
      color="#1f77b4",
      linewidth=2,
  )
  plt.plot(
      res_df["lambda"],
      res_df["mean_f1"],
      marker="^",
      label="Moyenne F1",
      color="#ff7f0e",
      linestyle="--",
  )
  plt.plot(
      res_df["lambda"],
      res_df["chunk_accuracy"] * 100,
      marker="d",
      label="Chunk Accuracy (TP+TN)",
      color="#9467bd",
      linestyle=":",
  )

  plt.title(
      f"Évolution des Métriques selon $\\lambda$ (Mode = '{stat_mode}', $\\gamma"
      f" = {gamma}$)",
      fontsize=12,
      pad=12,
  )
  plt.xlabel(
      "Valeur de $\\lambda$ (0.0 = Reranker pur, 1.0 = Lexical pur)",
      fontsize=11,
  )
  plt.ylabel("Score (%)", fontsize=11)
  plt.grid(True, linestyle="--", alpha=0.6)
  plt.legend(fontsize=11)
  plt.tight_layout()
  plt.show()

  return res_df










def run_gamma_grid_search(
    df,
    fixed_lambda=0.8,
    stat_mode="mean",
    gt_col="preannotation",
    gamma_start=0.5,
    gamma_end=1.2,
    step=0.05,
):
  gammas = np.arange(gamma_start, gamma_end + 1e-5, step)
  results = []

  print(f"\n{'='*95}")
  print(
      f"GRID SEARCH SUR GAMMA (Lambda fixé = {fixed_lambda:.2f} | Stat ="
      f" '{stat_mode}')"
  )
  print(f"{'='*95}")
  print(
      f"{'Gamma':<8} | {'Rappel %':<10} | {'Précision %':<12} | {'F1 %':<8} |"
      f" {'Exact %':<9} | {'TP':<6} | {'FP':<6} | {'FN':<6} | {'TN':<6}"
  )
  print(f"{'-'*95}")

  # Pré-calcul des scores pour accélérer l'exécution
  cached_scores = [
      compute_scores_for_sample(row, lambda_val=fixed_lambda)
      for _, row in df.iterrows()
  ]

  for g_val in gammas:
    g_val = round(g_val, 3)
    temp_col = []

    for (_, row), scores in zip(df.iterrows(), cached_scores):
      enriched = apply_adaptive_validation(
          row, scores, stat_mode=stat_mode, gamma=g_val
      )
      temp_col.append(enriched)

    df_temp = df.copy()
    df_temp["temp_eval_col"] = temp_col

    m = evaluate_chunk_classification_metrics(
        df_temp, gt_col=gt_col, pred_col="temp_eval_col"
    )
    m["gamma"] = g_val
    results.append(m)

    print(
        f"{g_val:<8.2f} | {m['mean_recall']:<10.2f} |"
        f" {m['mean_precision']:<12.2f} | {m['mean_f1']:<8.2f} |"
        f" {m['chunk_accuracy']*100:<9.2f} | {m['tp']:<6} | {m['fp']:<6} |"
        f" {m['fn']:<6} | {m['tn']:<6}"
    )

  print(f"{'='*95}\n")
  res_df = pd.DataFrame(results)

  # Tracé des courbes
  plt.figure(figsize=(10, 6))
  plt.plot(
      res_df["gamma"],
      res_df["mean_recall"],
      marker="o",
      label="Moyenne Rappel",
      color="#2ca02c",
      linewidth=2.5,
  )
  plt.plot(
      res_df["gamma"],
      res_df["mean_precision"],
      marker="s",
      label="Moyenne Précision",
      color="#1f77b4",
      linewidth=2,
  )
  plt.plot(
      res_df["gamma"],
      res_df["mean_f1"],
      marker="^",
      label="Moyenne F1",
      color="#ff7f0e",
      linestyle="--",
  )
  plt.plot(
      res_df["gamma"],
      res_df["chunk_accuracy"] * 100,
      marker="d",
      label="Chunk Accuracy (TP+TN)",
      color="#9467bd",
      linestyle=":",
  )

  plt.title(
      "Évolution des Métriques selon $\\gamma$ (Tolérance |"
      f" $\\lambda={fixed_lambda}$ | Mode='{stat_mode}')",
      fontsize=12,
      pad=12,
  )
  plt.xlabel("Valeur de $\\gamma$", fontsize=11)
  plt.ylabel("Score (%)", fontsize=11)
  plt.grid(True, linestyle="--", alpha=0.6)
  plt.legend(fontsize=11)
  plt.tight_layout()
  plt.show()

  return res_df
