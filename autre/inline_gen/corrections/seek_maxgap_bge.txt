def evaluate_per_snippet_metrics(df, gt_col='preannotation', pred_col='inline_processed', verbose=False):
    def parse_ids(val):
        if val is None: return set()
        if isinstance(val, str):
            try:
                parsed = ast.literal_eval(val)
                if isinstance(parsed, list): return set([int(x) for x in parsed])
                elif isinstance(parsed, (int, float)): return {int(parsed)}
            except: return set()
        if isinstance(val, (list, np.ndarray)): return set([int(x) for x in val])
        return set()

    def normalize_row(row_val):
        if row_val is None: return []
        if isinstance(row_val, str):
            try: return ast.literal_eval(row_val)
            except: return []
        if isinstance(row_val, np.ndarray): return row_val.tolist()
        if isinstance(row_val, list): return row_val
        return []

    total_snippets = 0
    sum_recall, sum_precision, sum_f1 = 0.0, 0.0, 0.0
    count_exact_match, count_count_match = 0, 0
    total_tp, total_fp, total_fn = 0, 0, 0

    for idx, row in df.iterrows():
        gt_list = normalize_row(row[gt_col])
        pred_list = normalize_row(row[pred_col])

        if not isinstance(gt_list, list) or not isinstance(pred_list, list):
            continue

        gt_map = {int(s['snippet_id']): parse_ids(s['prediction']) for s in gt_list if isinstance(s, dict) and 'snippet_id' in s and 'prediction' in s}
        pred_map = {int(s['snippet_id']): parse_ids(s['citations']) for s in pred_list if isinstance(s, dict) and 'snippet_id' in s and 'citations' in s}

        all_ids = set(gt_map.keys()) | set(pred_map.keys())

        for sid in all_ids:
            total_snippets += 1
            gt_set = gt_map.get(sid, set())
            pred_set = pred_map.get(sid, set())

            if len(gt_set) == 0 and len(pred_set) == 0:
                rec, prec, f1 = 1.0, 1.0, 1.0
                is_exact, is_count_correct = True, True
            else:
                tp = len(gt_set & pred_set)
                fp = len(pred_set - gt_set)
                fn = len(gt_set - pred_set)

                total_tp += tp
                total_fp += fp
                total_fn += fn

                prec = tp / (tp + fp) if (tp + fp) > 0 else 0.0
                rec = tp / (tp + fn) if (tp + fn) > 0 else 0.0
                f1 = 2 * (prec * rec) / (prec + rec) if (prec + rec) > 0 else 0.0

                is_exact = (gt_set == pred_set)
                is_count_correct = (len(gt_set) == len(pred_set))

            sum_recall += rec
            sum_precision += prec
            sum_f1 += f1

            if is_exact: count_exact_match += 1
            if is_count_correct: count_count_match += 1

    mean_r = (sum_recall / total_snippets) * 100 if total_snippets > 0 else 0.0
    mean_p = (sum_precision / total_snippets) * 100 if total_snippets > 0 else 0.0
    mean_f1 = (sum_f1 / total_snippets) * 100 if total_snippets > 0 else 0.0
    pct_exact = (count_exact_match / total_snippets) * 100 if total_snippets > 0 else 0.0
    pct_count = (count_count_match / total_snippets) * 100 if total_snippets > 0 else 0.0

    if verbose:
        print("\n" + "="*80)
        print(" RÉSUMÉ GLOBAL (MOYENNE SUR TOUS LES SNIPPETS)")
        print("="*80)
        print(f"Total Snippets analysés : {total_snippets}")
        print(f"Moyenne Rappel (Recall)       : {mean_r:.2f}%")
        print(f"Moyenne Précision (Precision) : {mean_p:.2f}%")
        print(f"Moyenne F1 Score              : {mean_f1:.2f}%")
        print(f"Snippets EXACT MATCH          : {pct_exact:.2f}%")
        print(f"Snippets COUNT MATCH          : {pct_count:.2f}%")

    return {
        "mean_recall": mean_r, "mean_precision": mean_p, "mean_f1": mean_f1,
        "pct_exact": pct_exact, "pct_count": pct_count, "total_snippets": total_snippets
    }

def run_seek_per_snippet(df, embed_model, w=1, k=3):
    results = []

    for idx, row in df.iterrows():
        contexts = row.get("contexts", [])
        inline_processed = row.get("inline_processed", [])
        
        if not contexts or not inline_processed:
            row_dict = row.to_dict()
            row_dict['seek_candidates'] = []
            results.append(row_dict)
            continue

        # 1. Découpage en sous-unités (phrases) et mapping des IDs
        all_sentences = []
        sentence_chunk_ids = []
        
        for chunk in contexts:
            c_id = int(chunk['id'])
            # Utilisation de ta fonction de split existante
            sentences = split_insurance_text(chunk['text'])
            all_sentences.extend(sentences)
            sentence_chunk_ids.extend([c_id] * len(sentences))

        N = len(all_sentences)
        if N == 0:
            row_dict = row.to_dict()
            row_dict['seek_candidates'] = []
            results.append(row_dict)
            continue

        # 2. Vectorisation (Phrases et Snippets)
        e_sentences = embed_model.encode(all_sentences, convert_to_tensor=True)
        snippet_texts = [s['text'] for s in inline_processed]
        e_snippets = embed_model.encode(snippet_texts, convert_to_tensor=True)

        # 3. Fenêtres contextuelles glissantes SEEK
        window_embeddings = []
        for i in range(N):
            start = max(0, i - w)
            end = min(N, i + w + 1)
            v_i = torch.mean(e_sentences[start:end], dim=0)
            window_embeddings.append(v_i)
        window_embeddings = torch.stack(window_embeddings)

        # 4. Similarité brute
        win_norm = torch.nn.functional.normalize(window_embeddings, p=2, dim=1)
        snip_norm = torch.nn.functional.normalize(e_snippets, p=2, dim=1)
        similarity_matrix = torch.mm(snip_norm.float(), win_norm.T.float()).cpu().numpy()

        # 5. Lissage temporel
        smoothed_matrix = np.zeros_like(similarity_matrix)
        half_k = k // 2
        for i in range(N):
            start = max(0, i - half_k)
            end = min(N, i + half_k + 1)
            smoothed_matrix[:, i] = np.mean(similarity_matrix[:, start:end], axis=1)

        # 6. Seuillage adaptatif (Elbow) par snippet et validation du chunk parent
        seek_candidates_list = []
        
        for s_idx, snippet in enumerate(inline_processed):
            scores_for_snippet = smoothed_matrix[s_idx]
            sorted_scores = np.sort(scores_for_snippet)[::-1]
            
            if len(sorted_scores) < 2:
                tau_elbow = sorted_scores[0] if len(sorted_scores) > 0 else 0
            else:
                gaps = sorted_scores[:-1] - sorted_scores[1:]
                elbow_idx = np.argmax(gaps)
                tau_elbow = sorted_scores[elbow_idx + 1]

            # Validation : si une phrase passe le seuil, le chunk parent est sélectionné
            pred_cited_set = set()
            for i, score in enumerate(scores_for_snippet):
                if score >= tau_elbow:
                    c_id = sentence_chunk_ids[i]
                    pred_cited_set.add(c_id)
                    
            seek_candidates_list.append({
                "snippet_id": snippet["snippet_id"],
                "candidates": list(pred_cited_set) # Liste des chunks parents validés
            })

        row_dict = row.to_dict()
        row_dict['seek_candidates'] = seek_candidates_list
        results.append(row_dict)

    return pd.DataFrame(results)





def run_bge_grid_search(df, bge_cache, thresholds=[0.5, 0.6, 0.7, 0.8, 0.9, 0.95]):
    results = []
    
    print(f"\n{'='*95}")
    print(f"{'GRID SEARCH BGE (ÉCHELLE SNIPPET) - CONSERVATEUR':^95}")
    print(f"{'='*95}")
    print(f"{'Seuil':<8} | {'Rappel %':<10} | {'Précis %':<10} | {'F1 %':<8} | {'Exact %':<9} | {'Count %':<9}")
    print(f"{'-'*95}")

    for thres in thresholds:
        temp_eval_col = []
        
        for row_idx, row in df.iterrows():
            inline_processed = row.get("inline_processed", [])
            seek_candidates = {sc["snippet_id"]: sc["candidates"] for sc in row.get("seek_candidates", [])}
            
            row_new_snippets = []
            
            for snippet in inline_processed:
                s_id = int(snippet["snippet_id"])
                
                # 1. Les citations DE BASE (Union conservatrice)
                raw_cits = snippet.get("citations", [])
                if isinstance(raw_cits, np.ndarray): base_cits = set(raw_cits.tolist())
                elif isinstance(raw_cits, list): base_cits = set(raw_cits)
                else: base_cits = set()
                
                final_cits = set(base_cits)
                
                # 2. Filtrage BGE sur les candidats SEEK
                candidates = seek_candidates.get(s_id, [])
                for c_id in candidates:
                    # Récupération du score dans le cache
                    score = bge_cache.get(row_idx, {}).get(s_id, {}).get(c_id, 0.0)
                    if score >= thres:
                        final_cits.add(c_id) # L'opération set() gère les doublons
                
                row_new_snippets.append({
                    "snippet_id": s_id,
                    "text": snippet.get("text", ""),
                    "citations": sorted(list(final_cits))
                })
                
            temp_eval_col.append(row_new_snippets)
            
        # Création d'un DF temporaire pour l'évaluation
        df_temp = df.copy()
        df_temp["temp_eval_col"] = temp_eval_col
        
        # 3. Évaluation via tes fonctions
        m_snip = evaluate_per_snippet_metrics(df_temp, gt_col='preannotation', pred_col='temp_eval_col', verbose=False)
        m_snip["threshold"] = thres
        results.append(m_snip)
        
        print(f"{thres:<8.2f} | {m_snip['mean_recall']:<10.2f} | {m_snip['mean_precision']:<10.2f} | "
              f"{m_snip['mean_f1']:<8.2f} | {m_snip['pct_exact']:<9.2f} | {m_snip['pct_count']:<9.2f}")

    return pd.DataFrame(results)
