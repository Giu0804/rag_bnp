def get_seek_candidates_for_row(row, embed_model, w=1, k=3):
    """
    Applique la méthode SEEK pour chaque snippet de la ligne.
    Retourne un dictionnaire avec les candidats (SEEK passés + citations de base conservées).
    """
    contexts = row.get("contexts", [])
    inline_processed = row.get("inline_processed", [])
    
    # 1. Préparation globale du contexte pour la ligne
    all_sentences = []
    sentence_chunk_ids = []
    unique_ids = []
    
    # Dictionnaire pour retrouver le texte complet du chunk si besoin pour BGE
    chunk_text_map = {} 
    
    for ctx in contexts:
        c_id = int(ctx["id"])
        if c_id not in unique_ids:
            unique_ids.append(c_id)
        
        chunk_text_map[c_id] = ctx["text"]
        sentences = split_insurance_text(ctx["text"])
        all_sentences.extend(sentences)
        sentence_chunk_ids.extend([c_id] * len(sentences))

    snippet_results = {}
    
    if not all_sentences or not inline_processed:
        return snippet_results

    # Vectorisation du contexte (une seule fois par ligne)
    e_sentences = embed_model.encode(all_sentences, convert_to_tensor=True)
    N = len(all_sentences)
    
    # Fenêtre glissante SEEK
    window_embeddings = []
    for i in range(N):
        start = max(0, i - w)
        end = min(N, i + w + 1)
        v_i = torch.mean(e_sentences[start:end], dim=0)
        window_embeddings.append(v_i)
        
    window_embeddings = torch.stack(window_embeddings)
    window_embeddings_norm = torch.nn.functional.normalize(window_embeddings, p=2, dim=1)

    # 2. Traitement par Snippet
    for item in inline_processed:
        sid = int(item["snippet_id"])
        snip_text = item.get("text", "")
        
        # Récupération sécurisée des citations de base (qu'on garde coûte que coûte)
        raw_cits = item.get("citations", item.get("citation", []))
        if isinstance(raw_cits, np.ndarray):
            base_cits = set(int(c) for c in raw_cits.tolist())
        elif isinstance(raw_cits, list):
            base_cits = set(int(c) for c in raw_cits)
        else:
            base_cits = set()

        if not snip_text.strip():
            # Si le texte est vide, on renvoie juste les citations de base
            snippet_results[sid] = {
                "text": snip_text,
                "base_cits": base_cits,
                "candidates": base_cits
            }
            continue

        # Vectorisation du snippet
        e_claim = embed_model.encode([snip_text], convert_to_tensor=True)
        e_claim_norm = torch.nn.functional.normalize(e_claim, p=2, dim=1)
        
        # Matrice de similarité brute
        sim_matrix = torch.mm(e_claim_norm.float(), window_embeddings_norm.T.float()).cpu().numpy()[0]
        
        # Temporal Smoothing
        smoothed_scores = np.zeros_like(sim_matrix)
        half_k = k // 2
        for i in range(N):
            start = max(0, i - half_k)
            end = min(N, i + half_k + 1)
            smoothed_scores[i] = np.mean(sim_matrix[start:end])

        # Dynamic Thresholding (Elbow)
        sorted_scores = np.sort(smoothed_scores)[::-1]
        if len(sorted_scores) < 2:
            tau_elbow = sorted_scores[0] if len(sorted_scores) > 0 else 0
        else:
            gaps = sorted_scores[:-1] - sorted_scores[1:]
            elbow_idx = np.argmax(gaps)
            tau_elbow = sorted_scores[elbow_idx + 1]

        # Identifier les chunks approuvés par SEEK
        seek_cited = set()
        for i, score in enumerate(smoothed_scores):
            if score >= tau_elbow:
                seek_cited.add(sentence_chunk_ids[i])

        # FUSION : Les candidats incluent toujours les base_cits
        final_candidates = base_cits.union(seek_cited)
        
        snippet_results[sid] = {
            "text": snip_text,
            "base_cits": base_cits,
            "candidates": final_candidates,
            "chunk_text_map": chunk_text_map # stocké pour l'étape BGE
        }
        
    return snippet_results






from tqdm import tqdm

def compute_and_cache_bge_scores(df, embed_model, bge_model, bge_tokenizer, batch_size=128):
    """
    1. Fait tourner SEEK sur chaque ligne.
    2. Prépare toutes les paires Snippet-Chunk candidats.
    3. Fait une inférence batchée avec BGE.
    4. Retourne un dictionnaire des scores complet.
    """
    all_pairs = []
    mapping = [] # (row_idx, snippet_id, chunk_id)
    
    # Stockage de la structure de base (pour le grid search plus tard)
    # structure : precomputed_data[row_idx][snippet_id] = { 'base_cits': set(), 'bge_scores': {chunk_id: score} }
    precomputed_data = {}

    print("Étape 1 : Passage de SEEK sur tout le dataset...")
    for idx, row in tqdm(df.iterrows(), total=len(df)):
        seek_res = get_seek_candidates_for_row(row, embed_model, w=1, k=3)
        precomputed_data[idx] = {}
        
        for sid, s_data in seek_res.items():
            precomputed_data[idx][sid] = {
                "text": s_data["text"],
                "base_cits": s_data["base_cits"],
                "bge_scores": {} # À remplir avec BGE
            }
            
            # Créer les paires pour BGE (seulement pour les candidats SEEK + base)
            for c_id in s_data["candidates"]:
                chunk_text = s_data["chunk_text_map"].get(c_id, "")
                if chunk_text:
                    all_pairs.append([s_data["text"], chunk_text])
                    mapping.append((idx, sid, c_id))

    print(f"Étape 2 : Inférence BGE sur {len(all_pairs)} paires...")
    all_scores = []
    for i in tqdm(range(0, len(all_pairs), batch_size)):
        batch_pairs = all_pairs[i : i + batch_size]
        
        inputs = bge_tokenizer(batch_pairs, padding=True, truncation=True, return_tensors='pt', max_length=512).to('cuda')
        
        with torch.no_grad():
            outputs = bge_model(**inputs)
            # Logits -> Probabilités (selon le modèle BGE exact, ajustez sigmoid vs softmax si besoin)
            logits = torch.sigmoid(outputs.logits.view(-1).float()).cpu().numpy()
            all_scores.extend(logits)

    print("Étape 3 : Mapping des scores...")
    for (row_idx, sid, chunk_id), score_val in zip(mapping, all_scores):
        # On sauvegarde le score BGE pour ce chunk précis
        precomputed_data[row_idx][sid]["bge_scores"][chunk_id] = float(np.round(score_val, 4))

    return precomputed_data







def evaluate_per_snippet_metrics(df, gt_col='preannotation', pred_col='inline_processed', verbose=False):

    def parse_ids(val):

        if val is None: return set()

        if isinstance(val, str):

                parsed = ast.literal_eval(val)

                if isinstance(parsed, list): return set([int(x) for x in parsed])

                elif isinstance(parsed, (int, float)): return {int(parsed)}

            except: return set()

        if isinstance(val, (list, np.ndarray)): return set([int(x) for x in val])

        return set()

 

    def normalize_row(row_val):

        if row_val is None: return []

        if isinstance(row_val, str):

            try: return ast.literal_eval(row_val)

            except: return []

        if isinstance(row_val, np.ndarray): return row_val.tolist()

        if isinstance(row_val, list): return row_val

        return []

 

    total_snippets = 0

    sum_recall = 0.0

    sum_precision = 0.0

    sum_f1 = 0.0

    count_exact_match = 0

    count_count_match = 0

   

    total_tp = 0

    total_fp = 0

    total_fn = 0

 

    for idx, row in df.iterrows():

        gt_list = normalize_row(row[gt_col])

        pred_list = normalize_row(row[pred_col])

 

        if not isinstance(gt_list, list) or not isinstance(pred_list, list):

            continue

 

        gt_map = {}

        for s in gt_list:

            if isinstance(s, dict) and 'snippet_id' in s and 'prediction' in s:

                gt_map[int(s['snippet_id'])] = parse_ids(s['prediction'])

 

        pred_map = {}

        for s in pred_list:

            if isinstance(s, dict) and 'snippet_id' in s and 'citations' in s:

                pred_map[int(s['snippet_id'])] = parse_ids(s['citations'])

 

        all_ids = set(gt_map.keys()) | set(pred_map.keys())

 

        for sid in all_ids:

            total_snippets += 1

            gt_set = gt_map.get(sid, set())

            pred_set = pred_map.get(sid, set())

 

            if len(gt_set) == 0 and len(pred_set) == 0:

                rec, prec, f1 = 1.0, 1.0, 1.0

                is_exact = True

                is_count_correct = True

            else:

                tp = len(gt_set & pred_set)

                fp = len(pred_set - gt_set)

                fn = len(gt_set - pred_set)

 

                total_tp += tp

                total_fp += fp

                total_fn += fn

 

                prec = tp / (tp + fp) if (tp + fp) > 0 else 0.0

                rec = tp / (tp + fn) if (tp + fn) > 0 else 0.0

                f1 = 2 * (prec * rec) / (prec + rec) if (prec + rec) > 0 else 0.0

 

                is_exact = (gt_set == pred_set)

                is_count_correct = (len(gt_set) == len(pred_set))

 

            sum_recall += rec

            sum_precision += prec

            sum_f1 += f1

 

            if is_exact: count_exact_match += 1

            if is_count_correct: count_count_match += 1

 

    mean_r = (sum_recall / total_snippets) * 100 if total_snippets > 0 else 0.0

    mean_p = (sum_precision / total_snippets) * 100 if total_snippets > 0 else 0.0

    mean_f1 = (sum_f1 / total_snippets) * 100 if total_snippets > 0 else 0.0

    pct_exact = (count_exact_match / total_snippets) * 100 if total_snippets > 0 else 0.0

    pct_count = (count_count_match / total_snippets) * 100 if total_snippets > 0 else 0.0

 

    if verbose:

        print("\n" + "="*80)

        print("📊 RÉSUMÉ GLOBAL (MOYENNE SUR TOUS LES SNIPPETS)")

        print("="*80)

        print(f"Total Snippets analysés : {total_snippets}")

        print(f"Moyenne Rappel (Recall)       : {mean_r:.2f}%")

        print(f"Moyenne Précision (Precision) : {mean_p:.2f}%")

        print(f"Moyenne F1 Score              : {mean_f1:.2f}%")

        print(f"Snippets EXACT MATCH          : {pct_exact:.2f}%")

        print(f"Snippets COUNT MATCH          : {pct_count:.2f}%")

 

    return {

        "mean_recall": mean_r,

        "mean_precision": mean_p,

        "mean_f1": mean_f1,

        "pct_exact": pct_exact,

        "pct_count": pct_count,

        "total_snippets": total_snippets,

        "tp": total_tp,

        "fp": total_fp,

        "fn": total_fn

    }

def calculate_aggregated_sample_metrics(df, gt_col='preannotation', pred_col='inline_processed'): 

    """ 

    Calcule les métriques en agrégeant toutes les citations/prédictions par sample (ensemble unique), 

    puis moyenne les résultats sur tous les samples. 

    """ 

 

    def parse_ids(val): 

        if val is None: return set() 

        if isinstance(val, str): 

            try: 

                parsed = ast.literal_eval(val) 

                if isinstance(parsed, list): return set([int(x) for x in parsed]) 

                elif isinstance(parsed, (int, float)): return {int(parsed)} 

            except: return set() 

        if isinstance(val, (list, np.ndarray)): return set([int(x) for x in val]) 

        return set() 

 

    def normalize_row(row_val): 

        if row_val is None: return [] 

        if isinstance(row_val, str): 

            try: return ast.literal_eval(row_val) 

            except: return [] 

        if isinstance(row_val, np.ndarray): return row_val.tolist() 

        if isinstance(row_val, list): return row_val 

        return [] 

 

    total_samples = 0 

    sum_recall = 0.0 

    sum_precision = 0.0 

    sum_f1 = 0.0 

    exact_match_count = 0 

    count_match_count = 0 

 

    for idx, row in df.iterrows(): 

        gt_raw = row[gt_col] 

        pred_raw = row[pred_col] 

 

        # Normalisation des listes de snippets 

        gt_list = normalize_row(gt_raw) 

        pred_list = normalize_row(pred_raw) 

 

        if not isinstance(gt_list, list) or not isinstance(pred_list, list): 

            continue 

 

        total_samples += 1 

 

        # --- AGGREGATION PAR SAMPLE (ENSEMBLE UNIQUE) --- 

        all_gt_citations = set() 

        all_pred_citations = set() 

 

        for s in gt_list: 

            if isinstance(s, dict) and 'snippet_id' in s and 'prediction' in s: 

                all_gt_citations.update(parse_ids(s['prediction'])) 

 

        for s in pred_list: 

            if isinstance(s, dict) and 'snippet_id' in s and 'citations' in s: 

                all_pred_citations.update(parse_ids(s['citations'])) 

 

        # Calcul des métriques sur l'ensemble agrégé de ce sample 

        if len(all_gt_citations) == 0 and len(all_pred_citations) == 0: 

            # Cas vide : 100% partout 

            rec, prec, f1 = 1.0, 1.0, 1.0 

            is_exact = True 

            is_count_correct = True 

        else: 

            tp = len(all_gt_citations & all_pred_citations) 

            fp = len(all_pred_citations - all_gt_citations) 

            fn = len(all_gt_citations - all_pred_citations) 

 

            prec = tp / (tp + fp) if (tp + fp) > 0 else 0.0 

            rec = tp / (tp + fn) if (tp + fn) > 0 else 0.0 

            f1 = 2 * (prec * rec) / (prec + rec) if (prec + rec) > 0 else 0.0 

 

            is_exact = (all_gt_citations == all_pred_citations) 

            is_count_correct = (len(all_gt_citations) == len(all_pred_citations)) 

 

        # Accumulation pour la moyenne globale 

        sum_recall += rec 

        sum_precision += prec 

        sum_f1 += f1 

 

        if is_exact: exact_match_count += 1 

        if is_count_correct: count_match_count += 1 

 

    # Affichage des résultats moyens 

    print(f"\n📊 AGGREGATED PER-SAMPLE METRICS (Mean of Sample-wise Aggregates):") 

    print("-" * 60) 

    print(f"Total Samples Evaluated: {total_samples}") 

 

    if total_samples > 0: 

        print(f"Exact Match Rate (Sample level): {(exact_match_count/total_samples)*100:.2f}%") 

        print(f"Count Match Rate (Sample level): {(count_match_count/total_samples)*100:.2f}%") 

        print(f"Average Recall:                   {(sum_recall/total_samples)*100:.2f}%") 

        print(f"Average Precision:                {(sum_precision/total_samples)*100:.2f}%") 

        print(f"Average F1 Score:                 {(sum_f1/total_samples)*100:.2f}%") 

    else: 

        print("No valid data found.")

 








def run_bge_grid_search_with_your_metrics(df, precomputed_data, thresholds=[0.1, 0.3, 0.5, 0.7, 0.9], gt_col='preannotation'):
    """
    Utilise le cache des scores BGE et TA fonction de métrique exacte pour trouver le meilleur seuil.
    """
    print(f"\n{'='*80}")
    print(f"{'GRID SEARCH BGE (SEEK FILTRÉ + CONSERVATION BASE) PAR SNIPPET':^80}")
    print(f"{'='*80}")
    print(f"{'Seuil':<8} | {'Rappel %':<10} | {'Précision %':<12} | {'F1 %':<8} | {'Exact %':<9}")
    print(f"{'-'*80}")
    
    best_f1 = 0
    best_thres = 0
    
    for thres in thresholds:
        temp_preds = []
        
        for idx, row in df.iterrows():
            row_cache = precomputed_data.get(idx, {})
            new_inline_processed = []
            
            # Reconstruire le format exact pour chaque snippet de la ligne
            for sid, s_data in row_cache.items():
                pred_cited = set()
                
                # 1. Ajout obligatoire des citations de base
                pred_cited.update(s_data["base_cits"])
                
                # 2. Ajout des candidats SEEK si leur score BGE >= thres
                for c_id, score in s_data["bge_scores"].items():
                    if score >= thres:
                        pred_cited.add(c_id)
                        
                # 3. Format attendu par ta fonction (dictionnaire)
                new_inline_processed.append({
                    "snippet_id": sid,
                    "citations": list(pred_cited)
                })
                
            temp_preds.append(new_inline_processed)
            
        # Créer un DataFrame temporaire pour l'évaluation
        df_temp = df.copy()
        df_temp['temp_pred_col'] = temp_preds
        
        # Appel de TA fonction de métrique
        metrics = evaluate_per_snippet_metrics(df_temp, gt_col=gt_col, pred_col='temp_pred_col', verbose=False)
        
        print(f"{thres:<8.2f} | {metrics['mean_recall']:<10.2f} | {metrics['mean_precision']:<12.2f} | "
              f"{metrics['mean_f1']:<8.2f} | {metrics['pct_exact']:<9.2f}")
        
        if metrics["mean_f1"] > best_f1:
            best_f1 = metrics["mean_f1"]
            best_thres = thres
            best_metrics = metrics
            
    print(f"{'='*80}")
    print(f"MEILLEUR SEUIL BGE : {best_thres} (F1: {best_f1:.2f}%)")
    return best_thres
