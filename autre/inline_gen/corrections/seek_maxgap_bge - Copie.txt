def run_seek_per_snippet(df, embed_model, w=1, k=3):
    results = []
    for idx, row in df.iterrows():
        contexts = row.get("contexts", [])
        inline_processed = row.get("inline_processed", [])
        
        if not contexts or not inline_processed:
            row_dict = row.copy().to_dict()
            row_dict['seek_candidates'] = []
            results.append(row_dict)
            continue

        all_sentences = []
        sentence_chunk_ids = []
        for chunk in contexts:
            c_id = int(chunk['id'])
            sentences = split_insurance_text(chunk['text'])
            all_sentences.extend(sentences)
            sentence_chunk_ids.extend([c_id] * len(sentences))

        N = len(all_sentences)
        if N == 0:
            continue

        e_sentences = embed_model.encode(all_sentences, convert_to_tensor=True)
        snippet_texts = [s['text'] for s in inline_processed]
        e_snippets = embed_model.encode(snippet_texts, convert_to_tensor=True)

        window_embeddings = []
        for i in range(N):
            start = max(0, i - w)
            end = min(N, i + w + 1)
            v_i = torch.mean(e_sentences[start:end], dim=0)
            window_embeddings.append(v_i)
        window_embeddings = torch.stack(window_embeddings)

        win_norm = torch.nn.functional.normalize(window_embeddings, p=2, dim=1)
        snip_norm = torch.nn.functional.normalize(e_snippets, p=2, dim=1)
        similarity_matrix = torch.mm(snip_norm.float(), win_norm.T.float()).cpu().numpy()

        smoothed_matrix = np.zeros_like(similarity_matrix)
        half_k = k // 2
        for i in range(N):
            start = max(0, i - half_k)
            end = min(N, i + half_k + 1)
            smoothed_matrix[:, i] = np.mean(similarity_matrix[:, start:end], axis=1)

        seek_candidates_list = []
        for s_idx, snippet in enumerate(inline_processed):
            scores = smoothed_matrix[s_idx]
            sorted_scores = np.sort(scores)[::-1]
            
            tau_elbow = 0
            if len(sorted_scores) >= 2:
                gaps = sorted_scores[:-1] - sorted_scores[1:]
                tau_elbow = sorted_scores[np.argmax(gaps) + 1]

            pred_cited_set = set()
            for i, score in enumerate(scores):
                if score >= tau_elbow:
                    pred_cited_set.add(sentence_chunk_ids[i])
                    
            seek_candidates_list.append({
                "snippet_id": int(snippet["snippet_id"]),
                "candidates": list(pred_cited_set)
            })

        row_dict = row.copy().to_dict()
        row_dict['seek_candidates'] = seek_candidates_list
        results.append(row_dict)

    return pd.DataFrame(results)


































def cache_bge_scores(df, tokenizer, model, batch_size=128):
    all_pairs = []
    mapping = [] 
    
    for row_idx, row in df.iterrows():
        contexts = {int(c['id']): c['text'] for c in row.get("contexts", [])}
        inline_processed = {int(s['snippet_id']): s['text'] for s in row.get("inline_processed", [])}
        
        for sc in row.get("seek_candidates", []):
            s_id = int(sc["snippet_id"])
            s_text = inline_processed.get(s_id, "")
            
            for c_id in sc["candidates"]:
                c_text = contexts.get(c_id, "")
                if s_text and c_text:
                    all_pairs.append([s_text, c_text])
                    mapping.append((row_idx, s_id, c_id))

    all_scores = []
    for i in tqdm(range(0, len(all_pairs), batch_size), desc="BGE Caching"):
        batch_pairs = all_pairs[i : i + batch_size]
        inputs = tokenizer(batch_pairs, padding=True, truncation=True, return_tensors='pt', max_length=512).to('cuda')
        with torch.no_grad():
            outputs = model(**inputs)
            logits = torch.sigmoid(outputs.logits.view(-1).float()).cpu().numpy()
            all_scores.extend(logits)

    cache = {}
    for (row_idx, s_id, c_id), score in zip(mapping, all_scores):
        if row_idx not in cache: cache[row_idx] = {}
        if s_id not in cache[row_idx]: cache[row_idx][s_id] = {}
        cache[row_idx][s_id][c_id] = float(np.round(score, 4))
        
    return cache

def calculate_aggregated_sample_metrics(df, gt_col='preannotation', pred_col='inline_processed'): 
    # [Ton code exact d'origine pour parser et normaliser...]
    def parse_ids(val):
        if val is None: return set()
        if isinstance(val, str):
            try:
                parsed = ast.literal_eval(val)
                if isinstance(parsed, list): return set([int(x) for x in parsed])
                elif isinstance(parsed, (int, float)): return {int(parsed)}
            except: return set()
        if isinstance(val, (list, np.ndarray)): return set([int(x) for x in val])
        return set()

    def normalize_row(row_val):
        if row_val is None: return []
        if isinstance(row_val, str):
            try: return ast.literal_eval(row_val)
            except: return []
        if isinstance(row_val, np.ndarray): return row_val.tolist()
        if isinstance(row_val, list): return row_val
        return []

    total_samples = 0 
    sum_recall, sum_precision, sum_f1 = 0.0, 0.0, 0.0 
    exact_match_count, count_match_count = 0, 0 

    for idx, row in df.iterrows(): 
        gt_list = normalize_row(row[gt_col]) 
        pred_list = normalize_row(row[pred_col]) 

        if not isinstance(gt_list, list) or not isinstance(pred_list, list): 
            continue 

        total_samples += 1 

        all_gt_citations = set() 
        all_pred_citations = set() 

        for s in gt_list: 
            if isinstance(s, dict) and 'snippet_id' in s and 'prediction' in s: 
                all_gt_citations.update(parse_ids(s['prediction'])) 

        for s in pred_list: 
            if isinstance(s, dict) and 'snippet_id' in s and 'citations' in s: 
                all_pred_citations.update(parse_ids(s['citations'])) 

        if len(all_gt_citations) == 0 and len(all_pred_citations) == 0: 
            rec, prec, f1 = 1.0, 1.0, 1.0 
            is_exact, is_count_correct = True, True 
        else: 
            tp = len(all_gt_citations & all_pred_citations) 
            fp = len(all_pred_citations - all_gt_citations) 
            fn = len(all_gt_citations - all_pred_citations) 

            prec = tp / (tp + fp) if (tp + fp) > 0 else 0.0 
            rec = tp / (tp + fn) if (tp + fn) > 0 else 0.0 
            f1 = 2 * (prec * rec) / (prec + rec) if (prec + rec) > 0 else 0.0 

            is_exact = (all_gt_citations == all_pred_citations) 
            is_count_correct = (len(all_gt_citations) == len(all_pred_citations)) 

        sum_recall += rec 
        sum_precision += prec 
        sum_f1 += f1 

        if is_exact: exact_match_count += 1 
        if is_count_correct: count_match_count += 1 

    mean_r = (sum_recall/total_samples)*100 if total_samples > 0 else 0.0
    mean_p = (sum_precision/total_samples)*100 if total_samples > 0 else 0.0
    mean_f1 = (sum_f1/total_samples)*100 if total_samples > 0 else 0.0
    pct_exact = (exact_match_count/total_samples)*100 if total_samples > 0 else 0.0
    pct_count = (count_match_count/total_samples)*100 if total_samples > 0 else 0.0

    return {
        "mean_recall": mean_r, "mean_precision": mean_p, "mean_f1": mean_f1,
        "pct_exact": pct_exact, "pct_count": pct_count
    }
