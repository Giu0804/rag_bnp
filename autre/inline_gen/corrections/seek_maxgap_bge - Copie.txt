import torch
import numpy as np
import pandas as pd
from tqdm import tqdm

def run_seek_per_snippet(df, embed_model, w=1, k=3):
    results = []
    for idx, row in df.iterrows():
        contexts = row.get("contexts", [])
        inline_processed = row.get("inline_processed", [])
        
        if not contexts or not inline_processed:
            row_dict = row.copy().to_dict()
            row_dict['seek_candidates'] = []
            results.append(row_dict)
            continue

        all_sentences = []
        sentence_chunk_ids = []
        for chunk in contexts:
            c_id = int(chunk['id'])
            sentences = split_insurance_text(chunk['text'])
            all_sentences.extend(sentences)
            sentence_chunk_ids.extend([c_id] * len(sentences))

        N = len(all_sentences)
        if N == 0:
            continue

        e_sentences = embed_model.encode(all_sentences, convert_to_tensor=True)
        snippet_texts = [s['text'] for s in inline_processed]
        e_snippets = embed_model.encode(snippet_texts, convert_to_tensor=True)

        window_embeddings = []
        for i in range(N):
            start = max(0, i - w)
            end = min(N, i + w + 1)
            v_i = torch.mean(e_sentences[start:end], dim=0)
            window_embeddings.append(v_i)
        window_embeddings = torch.stack(window_embeddings)

        win_norm = torch.nn.functional.normalize(window_embeddings, p=2, dim=1)
        snip_norm = torch.nn.functional.normalize(e_snippets, p=2, dim=1)
        similarity_matrix = torch.mm(snip_norm.float(), win_norm.T.float()).cpu().numpy()

        smoothed_matrix = np.zeros_like(similarity_matrix)
        half_k = k // 2
        for i in range(N):
            start = max(0, i - half_k)
            end = min(N, i + half_k + 1)
            smoothed_matrix[:, i] = np.mean(similarity_matrix[:, start:end], axis=1)

        seek_candidates_list = []
        for s_idx, snippet in enumerate(inline_processed):
            scores = smoothed_matrix[s_idx]
            sorted_scores = np.sort(scores)[::-1]
            
            tau_elbow = 0
            if len(sorted_scores) >= 2:
                gaps = sorted_scores[:-1] - sorted_scores[1:]
                tau_elbow = sorted_scores[np.argmax(gaps) + 1]

            pred_cited_set = set()
            for i, score in enumerate(scores):
                if score >= tau_elbow:
                    pred_cited_set.add(sentence_chunk_ids[i])
                    
            seek_candidates_list.append({
                "snippet_id": int(snippet["snippet_id"]),
                "candidates": list(pred_cited_set)
            })

        row_dict = row.copy().to_dict()
        row_dict['seek_candidates'] = seek_candidates_list
        results.append(row_dict)

    return pd.DataFrame(results)


































def cache_bge_scores(df, tokenizer, model, batch_size=128):
    all_pairs = []
    mapping = [] 
    
    for row_idx, row in df.iterrows():
        contexts = {int(c['id']): c['text'] for c in row.get("contexts", [])}
        inline_processed = {int(s['snippet_id']): s['text'] for s in row.get("inline_processed", [])}
        
        for sc in row.get("seek_candidates", []):
            s_id = int(sc["snippet_id"])
            s_text = inline_processed.get(s_id, "")
            
            for c_id in sc["candidates"]:
                c_text = contexts.get(c_id, "")
                if s_text and c_text:
                    all_pairs.append([s_text, c_text])
                    mapping.append((row_idx, s_id, c_id))

    all_scores = []
    for i in tqdm(range(0, len(all_pairs), batch_size), desc="BGE Caching"):
        batch_pairs = all_pairs[i : i + batch_size]
        inputs = tokenizer(batch_pairs, padding=True, truncation=True, return_tensors='pt', max_length=512).to('cuda')
        with torch.no_grad():
            outputs = model(**inputs)
            logits = torch.sigmoid(outputs.logits.view(-1).float()).cpu().numpy()
            all_scores.extend(logits)

    cache = {}
    for (row_idx, s_id, c_id), score in zip(mapping, all_scores):
        if row_idx not in cache: cache[row_idx] = {}
        if s_id not in cache[row_idx]: cache[row_idx][s_id] = {}
        cache[row_idx][s_id][c_id] = float(np.round(score, 4))
        
    return cache


























def run_bge_grid_search_with_retention(df, bge_cache, thresholds):
    results = []
    print(f"\n{'='*95}\n{'GRID SEARCH BGE (AVEC CONSERVATION)':^95}\n{'='*95}")
    
    for thres in thresholds:
        temp_eval_col = []
        for row_idx, row in df.iterrows():
            row_snippets = []
            seek_candidates = {sc["snippet_id"]: sc["candidates"] for sc in row.get("seek_candidates", [])}
            
            for snippet in row.get("inline_processed", []):
                s_id = int(snippet["snippet_id"])
                
                # Conservation des citations initiales
                raw_cits = snippet.get("citations", snippet.get("citation", []))
                if isinstance(raw_cits, np.ndarray): base_cits = set(raw_cits.tolist())
                elif isinstance(raw_cits, list): base_cits = set(raw_cits)
                else: base_cits = set()
                
                final_cits = set(base_cits)
                
                # Ajout conditionnel des candidats SEEK validés par BGE
                for c_id in seek_candidates.get(s_id, []):
                    score = bge_cache.get(row_idx, {}).get(s_id, {}).get(c_id, 0.0)
                    if score >= thres:
                        final_cits.add(c_id)
                
                row_snippets.append({
                    "snippet_id": s_id,
                    "text": snippet.get("text", ""),
                    "citations": sorted(list(final_cits))
                })
            temp_eval_col.append(row_snippets)
            
        df_temp = df.copy()
        df_temp["temp_eval_col"] = temp_eval_col
        
        m_snip = evaluate_per_snippet_metrics(df_temp, gt_col='preannotation', pred_col='temp_eval_col', verbose=False)
        m_snip["threshold"] = thres
        results.append(m_snip)
        print(f"Seuil {thres:<4} | Rappel: {m_snip['mean_recall']:.2f}% | Prec: {m_snip['mean_precision']:.2f}% | F1: {m_snip['mean_f1']:.2f}%")

    return pd.DataFrame(results)
