def safe_parse_ids(val):
    """Ta fonction robuste pour parser les IDs de citations."""
    if val is None: return set()
    if isinstance(val, str):
        try:
            parsed = ast.literal_eval(val)
            if isinstance(parsed, list): return set([int(x) for x in parsed])
            elif isinstance(parsed, (int, float)): return {int(parsed)}
        except: return set()
    if isinstance(val, (list, np.ndarray)): return set([int(x) for x in val])
    return set()



def compute_advanced_scores_for_sample(row, lambda_val):
    """
    Calcule le score hybride S = lambda * BM25_norm + (1 - lambda) * ROUGE-L
    pour chaque snippet par rapport à chaque document du contexte.
    """
    contexts = row.get("contexts", [])
    inline_processed = row.get("inline_processed", [])
    
    if not contexts or not inline_processed:
        return {}

    # 1. Préparation du corpus pour BM25 (chaque doc est tokenisé en liste de mots)
    doc_tokens_list = []
    doc_ids = []
    for ctx in contexts:
        d_id = int(ctx["id"])
        tokens = re.findall(r"\b\w+\b", str(ctx["text"]).lower())
        doc_tokens_list.append(tokens)
        doc_ids.append(d_id)
        
    bm25 = BM25Okapi(doc_tokens_list)
    
    # Initialisation du scorer ROUGE-L
    scorer = rouge_scorer.RougeScorer(['rougeL'], use_stemmer=True)

    snippet_scores = {} # {snippet_id: {doc_id: score}}

    for item in inline_processed:
        sid = int(item["snippet_id"])
        s_text = str(item.get("text", ""))
        s_tokens = re.findall(r"\b\w+\b", s_text.lower())
        
        if not s_tokens:
            snippet_scores[sid] = {d_id: 0.0 for d_id in doc_ids}
            continue

        # --- A. Calcul des scores bruts BM25 ---
        raw_bm25_scores = bm25.get_scores(s_tokens)
        
        # Normalisation Min-Max des scores BM25 pour ce snippet (évite les divisions par zéro)
        min_b = np.min(raw_bm25_scores)
        max_b = np.max(raw_bm25_scores)
        if max_b - min_b > 1e-5:
            norm_bm25 = (raw_bm25_scores - min_b) / (max_b - min_b)
        else:
            norm_bm25 = np.zeros_like(raw_bm25_scores)

        # --- B. Calcul des scores ROUGE-L et Fusion ---
        snippet_scores[sid] = {}
        for idx, d_id in enumerate(doc_ids):
            b_score = float(norm_bm25[idx])
            
            # Calcul ROUGE-L F1 entre le snippet et le texte du doc
            doc_text = str(contexts[idx].get("text", ""))
            rouge_res = scorer.score(doc_text, s_text)
            r_score = float(rouge_res['rougeL'].fmeasure)
            
            # Équation hybride combinant Fond (BM25) et Forme (ROUGE-L)
            s_ij = (lambda_val * b_score) + ((1.0 - lambda_val) * r_score)
            snippet_scores[sid][d_id] = float(np.round(s_ij, 4))

    return snippet_scores




def apply_advanced_adaptive_validation(row, snippet_scores, stat_mode="max", gamma=0.85, fallback_tau=0.30):
    inline_processed = row.get("inline_processed", [])
    enriched_snippets = []
    
    for item in inline_processed:
        sid = int(item["snippet_id"])
        base_cits = safe_parse_ids(item.get("citations", item.get("citation", [])))
        scores_for_snip = snippet_scores.get(sid, {})

        # 1. Calcul du seuil adaptatif selon tes stats (mean, max, min)
        if len(base_cits) > 0:
            initial_scores = [scores_for_snip.get(int(c), 0.0) for c in base_cits]
            
            if stat_mode == "mean":
                ref_stat = float(np.mean(initial_scores))
            elif stat_mode == "max":
                ref_stat = float(np.max(initial_scores))
            elif stat_mode == "min":
                ref_stat = float(np.min(initial_scores))
            else:
                raise ValueError(f"stat_mode inconnu: {stat_mode}")
                
            adaptive_tau = gamma * ref_stat
        else:
            adaptive_tau = fallback_tau

        # 2. UNION CONSERVATRICE : On part des citations de base
        new_cits = set(base_cits)
        
        # 3. Ajout des nouveaux documents qui dépassent le seuil adaptatif
        for d_id, score in scores_for_snip.items():
            if score >= adaptive_tau:
                new_cits.add(d_id)

        enriched_snippets.append({
            "snippet_id": sid,
            "text": item.get("text", ""),
            "citations": sorted(list(new_cits))
        })

    return enriched_snippets

def calculate_aggregated_sample_metrics(df, gt_col='preannotation', pred_col='inline_processed', verbose=False):
    """
    Calcule les métriques en agrégeant toutes les citations/prédictions par sample (ensemble unique),
    puis retourne un dictionnaire propre pour le Grid Search.
    """
    total_samples = 0 
    sum_recall, sum_precision, sum_f1 = 0.0, 0.0, 0.0 
    exact_match_count, count_match_count = 0, 0 

    for idx, row in df.iterrows(): 
        gt_raw = row.get(gt_col) 
        pred_raw = row.get(pred_col) 

        # Utilisation de ta fonction de parsing sécurisée
        gt_list = normalize_row_eval(gt_raw) 
        pred_list = normalize_row_eval(pred_raw) 

        if not isinstance(gt_list, list) or not isinstance(pred_list, list): 
            continue 

        total_samples += 1 

        all_gt_citations = set() 
        all_pred_citations = set() 

        for s in gt_list: 
            if isinstance(s, dict) and 'snippet_id' in s and 'prediction' in s: 
                all_gt_citations.update(safe_parse_ids(s['prediction'])) 

        for s in pred_list: 
            if isinstance(s, dict) and 'snippet_id' in s and 'citations' in s: 
                all_pred_citations.update(safe_parse_ids(s['citations'])) 

        if len(all_gt_citations) == 0 and len(all_pred_citations) == 0: 
            rec, prec, f1 = 1.0, 1.0, 1.0 
            is_exact, is_count_correct = True, True 
        else: 
            tp = len(all_gt_citations & all_pred_citations) 
            fp = len(all_pred_citations - all_gt_citations) 
            fn = len(all_gt_citations - all_gt_citations) # (corrigé ici aussi)

            prec = tp / (tp + fp) if (tp + fp) > 0 else 0.0 
            rec = tp / (tp + len(all_gt_citations - all_pred_citations)) if len(all_gt_citations) > 0 else 1.0 
            f1 = 2 * (prec * rec) / (prec + rec) if (prec + rec) > 0 else 0.0 

            is_exact = (all_gt_citations == all_pred_citations) 
            is_count_correct = (len(all_gt_citations) == len(all_pred_citations)) 

        sum_recall += rec 
        sum_precision += prec 
        sum_f1 += f1 

        if is_exact: exact_match_count += 1 
        if is_count_count_correct := is_count_correct: count_match_count += (1 if is_count_correct else 0)

    mean_r = (sum_recall / total_samples) * 100 if total_samples > 0 else 0.0
    mean_p = (sum_precision / total_samples) * 100 if total_samples > 0 else 0.0
    mean_f1 = (sum_f1 / total_samples) * 100 if total_samples > 0 else 0.0
    pct_exact = (exact_match_count / total_samples) * 100 if total_samples > 0 else 0.0
    pct_count = (count_match_count / total_samples) * 100 if total_samples > 0 else 0.0

    if verbose:
        print(f"\n📊 AGGREGATED PER-SAMPLE METRICS:")
        print("-" * 50)
        print(f"Total Samples Evaluated: {total_samples}")
        print(f"Average Recall (Sample):    {mean_r:.2f}%")
        print(f"Average Precision (Sample): {mean_p:.2f}%")
        print(f"Average F1 Score (Sample):  {mean_f1:.2f}%")

    # RETOURNE ENFIN UN DICTIONNAIRE EXPLOITABLE
    return {
        "mean_recall": mean_r,
        "mean_precision": mean_p,
        "mean_f1": mean_f1,
        "pct_exact": pct_exact,
        "pct_count": pct_count,
        "total_samples": total_samples
    }

def normalize_row_eval(val):
    if val is None: return []
    if isinstance(val, str):
        try: return ast.literal_eval(val)
        except: return []
    if isinstance(val, np.ndarray): return val.tolist()
    if isinstance(val, list): return val
    return []

    for row_idx, row in df.iterrows():
        contexts = row.get("contexts", [])
        inline_processed = row.get("inline_processed", [])
        
        if not contexts or not inline_processed:
            continue
            
        row_cache = {}
        for item in inline_processed:
            sid = int(item["snippet_id"])
            s_text = str(item.get("text", ""))
            s_tokens = set(re.findall(r"\b\w+\b", s_text.lower()))
            
            row_cache[sid] = {}
            for ctx in contexts:
                d_id = int(ctx['id'])
                doc_text = str(ctx['text'])
                doc_tokens = set(re.findall(r"\b\w+\b", doc_text.lower()))
                
                # --- A. Calcul de l'Intersection Asymétrique ---
                if len(s_tokens) == 0:
                    intersect_score = 0.0
                else:
                    common = s_tokens & doc_tokens
                    intersect_score = len(common) / len(s_tokens)
                
                # --- B. Calcul ROUGE-L F1 ---
                rouge_res = scorer.score(doc_text, s_text)
                r_score = float(rouge_res['rougeL'].fmeasure)
                
                row_cache[sid][d_id] = {
                    'intersection': float(np.round(intersect_score, 4)),
                    'rouge': float(np.round(r_score, 4))
                }
        cache[row_idx] = row_cache
        
    print(">> Cache Intersection + ROUGE-L généré avec succès !")
    return cache
