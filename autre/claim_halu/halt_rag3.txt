def parse_atomic_claims(atomic_claims) -> List[str]:
    if isinstance(atomic_claims, list):
        return [str(c).strip() for c in atomic_claims if str(c).strip()]
    if isinstance(atomic_claims, str):
        return [f.strip("- *").strip() for f in atomic_claims.split("\n") if f.strip()]
    return []

def get_windows(text: str, window_size: int = 320, stride: int = 320) -> List[str]:
    """
    Découpe le texte en utilisant une fenêtre non chevauchante de 320 tokens avec un pas de 320[cite: 1].
    Une approche par mots est utilisée ici comme proxy léger pour les tokens.
    """
    words = text.split()
    if not words:
        return [""]
    
    windows = []
    for i in range(0, len(words), stride):
        window = " ".join(words[i : i + window_size])
        windows.append(window)
    return windows

def compute_lexical_features(claim: str, chunk_window: str) -> dict:
    """
    Calcule les features lexicales légères : longueurs, ratios, ROUGE-L et Jaccard[cite: 1].
    """
    scorer = rouge_scorer.RougeScorer(['rougeL'], use_stemmer=True)
    
    # Longueurs de séquence[cite: 1]
    len_claim = len(claim.split())
    len_chunk = len(chunk_window.split())
    
    # Ratios de longueur[cite: 1]
    length_ratio = len_claim / max(len_chunk, 1)
    
    # Similarité Jaccard[cite: 1]
    set_claim = set(claim.lower().split())
    set_chunk = set(chunk_window.lower().split())
    intersection = len(set_claim.intersection(set_chunk))
    union = len(set_claim.union(set_chunk))
    jaccard = intersection / max(union, 1)
    
    # Chevauchement ROUGE-L[cite: 1]
    rouge_l_score = scorer.score(chunk_window, claim)['rougeL'].fmeasure
    
    return {
        "len_claim": len_claim,
        "len_chunk": len_chunk,
        "length_ratio": length_ratio,
        "jaccard_sim": jaccard,
        "rouge_l": rouge_l_score
    }





# =========================================================
# 1. SCORING CLAIM-LEVEL — NLI (Avec Fenêtrage)
# =========================================================

def get_claim_level_nli_scores(
    df: pd.DataFrame, nli_model, batch_size: int = 64, 
    claims_col: str = "atomic_claims", chunks_col: str = "retrieved_chunks"
) -> Dict[Tuple, Dict]:
    
    all_pairs, all_claims, sample_indices, chunk_ids, window_texts = [], [], [], [], []

    for idx, row in df.iterrows():
        claims = parse_atomic_claims(row[claims_col])
        chunks = row.get(chunks_col, []) if isinstance(row, dict) else row[chunks_col]
        if not claims or not chunks:
            continue
            
        for claim in claims:
            for chunk in chunks:
                # Application du fenêtrage sur le chunk
                windows = get_windows(chunk["text"], window_size=320, stride=320)
                for window in windows:
                    all_pairs.append({"text": window, "text_pair": claim})
                    all_claims.append(claim)
                    sample_indices.append(idx)
                    chunk_ids.append(chunk["id"])
                    window_texts.append(window)

    if not all_pairs:
        return {}

    nli_best: Dict[Tuple, Dict] = {}

    for i in tqdm(range(0, len(all_pairs), batch_size), desc="NLI inference"):
        batch = all_pairs[i : i + batch_size]
        results = nli_model(batch, top_k=None, batch_size=batch_size)

        for j, pair_results in enumerate(results):
            entail, contradict, neutral = 0.0, 0.0, 0.0
            items = pair_results if isinstance(pair_results, list) else [pair_results]
            for item in items:
                lbl = str(item.get("label", "")).upper()
                score = item.get("score", 0.0)
                if "ENTAIL" in lbl: entail = score
                elif "CONTRADICT" in lbl: contradict = score
                elif "NEUTRAL" in lbl: neutral = score

            global_idx = i + j
            key = (sample_indices[global_idx], all_claims[global_idx])
            
            # Calcul des features lexicales sur la fenêtre courante
            lex_feats = compute_lexical_features(all_claims[global_idx], window_texts[global_idx])
            
            # Stratégie Max Pooling[cite: 1]
            prev = nli_best.get(key, {"entail": -1.0})
            if entail > prev.get("entail", -1.0):
                nli_best[key] = {
                    "entail": entail,
                    "contradict": contradict,
                    "neutral": neutral,
                    "best_chunk_id": chunk_ids[global_idx],
                    "best_chunk_text": window_texts[global_idx],
                    **lex_feats # On stocke les features lexicales de la meilleure fenêtre
                }

    return nli_best

# 2. SCORING CLAIM-LEVEL — RERANKER (Avec Fenêtrage)
# =========================================================

def get_claim_level_reranker_scores(
    df: pd.DataFrame, reranker_model, tokenizer, batch_size: int = 128,
    claims_col: str = "atomic_claims", chunks_col: str = "retrieved_chunks", device: str = "cuda"
) -> Dict[Tuple, Dict]:
    
    all_pairs, all_claims, sample_indices, chunk_ids, window_texts = [], [], [], [], []

    for idx, row in df.iterrows():
        claims = parse_atomic_claims(row[claims_col])
        chunks = row.get(chunks_col, []) if isinstance(row, dict) else row[chunks_col]
        if not claims or not chunks:
            continue
            
        for claim in claims:
            for chunk in chunks:
                windows = get_windows(chunk["text"], window_size=320, stride=320)
                for window in windows:
                    all_pairs.append([claim, window])
                    all_claims.append(claim)
                    sample_indices.append(idx)
                    chunk_ids.append(chunk["id"])
                    window_texts.append(window)

    if not all_pairs:
        return {}

    rerank_best: Dict[Tuple, Dict] = {}

    for i in tqdm(range(0, len(all_pairs), batch_size), desc="Reranker inference"):
        batch_pairs = all_pairs[i : i + batch_size]
        inputs = tokenizer(batch_pairs, padding=True, truncation=True, return_tensors="pt", max_length=512).to(device)
        
        with torch.no_grad():
            probs = torch.sigmoid(reranker_model(**inputs).logits.view(-1).float()).cpu().numpy()

        for j, score_val in enumerate(probs):
            global_idx = i + j
            key = (sample_indices[global_idx], all_claims[global_idx])
            
            # Stratégie Max Pooling[cite: 1]
            prev = rerank_best.get(key, {"rerank_score": -1.0})
            if score_val > prev["rerank_score"]:
                rerank_best[key] = {
                    "rerank_score": float(np.round(score_val, 4)),
                    "best_chunk_id": chunk_ids[global_idx],
                    "best_chunk_text": window_texts[global_idx],
                }

    return rerank_best

# 3. CONSTRUCTEUR DE TABLE DE FEATURES
# =========================================================

def build_feature_table(nli_scores: Dict[Tuple, Dict], rerank_scores: Dict[Tuple, Dict]) -> pd.DataFrame:
    rows = []
    for key, nli_s in nli_scores.items():
        rerank_s = rerank_scores.get(key, {"rerank_score": 0.0, "best_chunk_id": None, "best_chunk_text": ""})
        
        rows.append({
            "row_idx": key[0],
            "claim": key[1],
            "nli_entail": nli_s["entail"],
            "nli_contradict": nli_s["contradict"],
            "nli_neutral": nli_s.get("neutral", 0.0),
            "rerank_score": rerank_s["rerank_score"],
            
            # Nouvelles features lexicales[cite: 1]
            "len_claim": nli_s.get("len_claim", 0),
            "len_chunk": nli_s.get("len_chunk", 0),
            "length_ratio": nli_s.get("length_ratio", 0.0),
            "jaccard_sim": nli_s.get("jaccard_sim", 0.0),
            "rouge_l": nli_s.get("rouge_l", 0.0),
            
            "nli_best_chunk_text": nli_s.get("best_chunk_text", ""),
            "rerank_best_chunk_text": rerank_s.get("best_chunk_text", ""),
        })
    return pd.DataFrame(rows)

# 5. ENTRAINEMENT, CALIBRATION ET ABSTENTION
# =========================================================

def train_calibrated_meta_classifier(
    calibration_df: pd.DataFrame, feature_cols: List[str] = FEATURE_COLS, cv_folds: int = 5
):
    X = calibration_df[feature_cols].values
    y = calibration_df["label"].values
    groups = calibration_df["row_idx"].values

    # Classifieur de base avec poids équilibrés[cite: 1]
    base_clf = LogisticRegression(class_weight="balanced", max_iter=1000)
    
    # Protocole d'entraînement out-of-fold (OOF) à 5 plis avec Isotonic Regression[cite: 1]
    gkf = GroupKFold(n_splits=cv_folds)
    calibrated_clf = CalibratedClassifierCV(
        estimator=base_clf, 
        method="isotonic",
        cv=gkf
    )
    
    # L'entraînement ajuste la calibration post-hoc sur les prédictions OOF
    calibrated_clf.fit(X, y, groups=groups)
    
    # Génération des probabilités calibrées sur les données d'entraînement 
    # (En production, ceci est fait sur un set de validation OOF)
    y_probs = calibrated_clf.predict_proba(X)[:, 1]
    
    return calibrated_clf, y_probs, y

def optimize_threshold(y_true, y_probs, min_precision=0.70):
    """
    Trouve le seuil qui maximise le score F1 sous la contrainte Precision >= 0.70[cite: 1].
    """
    precisions, recalls, thresholds = precision_recall_curve(y_true, y_probs)
    best_f1, optimal_threshold = 0, 0.5
    
    for p, r, t in zip(precisions, recalls, thresholds):
        if p >= min_precision:
            f1 = 2 * (p * r) / (p + r) if (p + r) > 0 else 0
            if f1 > best_f1:
                best_f1 = f1
                optimal_threshold = t
                
    return optimal_threshold

def apply_abstention_policy(y_probs, threshold, margin=0.10):
    """
    Implémente un mécanisme de rejet (abstention) autour du seuil d'incertitude[cite: 1].
    Retourne -1 pour les cas où le modèle choisit de s'abstenir.
    """
    predictions = []
    for p in y_probs:
        if (threshold - margin) < p < (threshold + margin):
            predictions.append(-1) # Abstention
        else:
            predictions.append(1 if p >= threshold else 0)
    return np.array(predictions)
