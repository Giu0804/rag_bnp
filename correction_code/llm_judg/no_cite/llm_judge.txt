import asyncio
import time
import os
import requests
import pandas as pd
from typing import List, Dict, Any, Optional
from openai import AsyncOpenAI

class LLMJudge:
    """
    Unified LLM Judge for evaluating context usage in answers (No Citation).
    Supports:
    - 'llmas': Remote LLM-as-a-Service
    - 'vllm': Local VLLM server
    - 'qwen': Local Qwen server (Streaming + Reasoning logic)
    """
    def __init__(
        self,
        judge_type: str = "vllm",
        base_url: Optional[str] = None,
        model_name: Optional[str] = None,
        api_key: Optional[str] = None,
        supports_xhigh: bool = False
    ):
        self.judge_type = judge_type
        self.supports_xhigh = supports_xhigh
        self.model_name = model_name
        self.base_url = base_url

        # --- GESTION DE LA CLÉ API ---
        if api_key is None:
            if judge_type in ["vllm", "qwen"]:
                try:
                    # Remplacement localhost si possible par "EMPTY" ou gestion locale
                    api_key = requests.get("http://localhost:8899/access-token").text
                except Exception as e:
                    print(f"Warning: Could not fetch API key for {judge_type}, defaulting to 'EMPTY'. Error: {e}")
                    api_key = "EMPTY"
            else:
                api_key = os.getenv("LLM_API_KEY", "EMPTY")

        # --- GESTION DU CERTIFICAT SSL (LLMAS) ---
        if judge_type in ["qwen", "llmas"]:
            ca_bundle = os.getenv("REQUESTS_CA_BUNDLE")
            if ca_bundle:
                os.environ["SSL_CERT_FILE"] = ca_bundle

        self.client = AsyncOpenAI(base_url=base_url, api_key=api_key)

    async def _get_model_name(self) -> str:
        if self.model_name:
            return self.model_name
        models = await self.client.models.list()
        if not models.data:
            raise ValueError("No models available on the endpoint.")
        self.model_name = models.data[0].id
        return self.model_name

    async def judge_statement(
        self,
        question: str,
        statement: str,
        context_chunks: List[Dict[str, Any]],
        system_prompt: str,
        user_prompt_template: str,
        temperature: float = 0.0,
        max_tokens: int = 512
    ) -> Dict[str, Any]:
        
        # Formatage du contexte
        context_text = "\n".join([f"ID: {c.get('id', i)} | Paragraph: {c.get('text', '')}" for i, c in enumerate(context_chunks)])
        user_prompt = user_prompt_template.format(
            context_text=context_text,
            answer_text=statement
        )

        model_name = await self._get_model_name()

        kwargs = {
            "model": model_name,
            "messages": [
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": user_prompt}
            ],
            "temperature": temperature,
            "max_tokens": max_tokens
        }

        # --- SPÉCIFICITÉS PAR TYPE DE SERVEUR ---
        if self.judge_type == "qwen":
            kwargs["stream"] = True
            kwargs["stream_options"] = {"include_usage": True}
            kwargs["extra_body"] = {"chat_template_kwargs": {"enable_thinking": True}}
            kwargs["reasoning_effort"] = "xhigh" if self.supports_xhigh else "high"
        elif self.judge_type == "llmas":
            kwargs["extra_body"] = {"reasoning_effort": "high"}
        elif self.judge_type == "vllm":
            kwargs["reasoning_effort"] = "high"

        # --- APPEL API ---
        start_time = time.time()
        try:
            if self.judge_type == "qwen":
                # Logique de streaming asynchrone spécifique
                response = await self.client.chat.completions.create(**kwargs)
                answer_content = ""
                async for chunk in response:
                    if chunk.choices:
                        delta = chunk.choices[0].delta
                        if hasattr(delta, "content") and delta.content:
                            answer_content += delta.content
                prediction = answer_content.strip()
            else:
                # Logique classique
                response = await self.client.chat.completions.create(**kwargs)
                prediction = response.choices[0].message.content.strip()

            latency = time.time() - start_time
            return {"prediction": prediction, "latency": latency, "model": model_name}

        except Exception as e:
            latency = time.time() - start_time
            return {"prediction": f"Error: {str(e)}", "latency": latency, "model": model_name, "error": str(e)}


async def run_judge_pipeline(
    data: pd.DataFrame,
    judge: LLMJudge,
    system_prompt: str,
    user_prompt_template: str,
    batch_size: int = 16,
    temperature: float = 0.0,
    max_tokens: int = 512
) -> pd.DataFrame:
    
    semaphore = asyncio.Semaphore(batch_size)

    async def process_row(sample_idx, sample_row):
        async with semaphore:
            question = sample_row['question']
            contexts = sample_row['contexts']
            statement = sample_row.get('generated_answer', '')

            if not isinstance(statement, str) or not statement.strip():
                return {
                    "sample_id": sample_row['sample_id'], 
                    "prediction": "Error: No generated_answer found", 
                    "latency": 0.0, 
                    "model": "N/A"
                }

            result = await judge.judge_statement(
                question=question,
                statement=statement,
                context_chunks=contexts,
                system_prompt=system_prompt,
                user_prompt_template=user_prompt_template,
                temperature=temperature,
                max_tokens=max_tokens
            )

            print(f"[{judge.judge_type}] Processed sample {sample_row['sample_id']} | Latency: {result['latency']:.2f}s")
            
            return {
                "sample_id": sample_row['sample_id'],
                "prediction": result['prediction'],
                "latency": result['latency'],
                "model": result['model']
            }

    tasks = [process_row(idx, row) for idx, row in data.iterrows()]
    results = await asyncio.gather(*tasks)
    
    # Sécurité : Tri par sample_id pour s'assurer d'un merge parfait par la suite
    df_results = pd.DataFrame(results).sort_values("sample_id").reset_index(drop=True)
    return df_results
